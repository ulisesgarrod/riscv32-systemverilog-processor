# Fibonacci for the pipelined processor

auipc x3, 65536
nop                 # Bubble: wait for x3 to be updated
nop                 # Second bubble
nop

lw x7, 0(x3)

addi x8, x3, 8      # Pointer 1
addi x9, x3, 4      # Pointer 2

addi x10, x0, 1     # x10 = 1
addi x11, x0, 0     # x11 = 0

addi x12, x0, 0
addi x13, x0, 2
nop
nop
nop

sw x11, 0(x9)       # Store 0
sw x10, 0(x8)       # Store 1
nop
nop
nop

LOOP:
lw x10, 0(x9)
lw x11, 0(x8)
nop                 # Wait for load writeback
nop
nop

add x12, x11, x10

nop                 # Wait for x12 writeback
nop
nop

sw x12, 4(x8)

addi x8, x8, 4
addi x9, x9, 4
addi x13, x13, 1
nop                 # Wait for x13 writeback
nop
nop

bne x7, x13, LOOP
nop
nop
nop

.data
.word 10
